```typescript
// Resolve detail cost: MAX(materialCost, unitPrice * area) per architecture decision
function resolveDetailCost(detail: { materialCost: unknown; unitPrice: unknown; area: unknown }): number {
  const mat = Number(detail.materialCost ?? 0);
  const unitTimesArea = Number(detail.unitPrice ?? 0) * Number(detail.area ?? 0);
  return Math.max(mat, unitTimesArea);
}
```

Aggregate recomputation helper:
```typescript
async function recomputeCostItemAggregates(db: PrismaClient, costItemId: string) {
  const details = await db.costItemDetail.findMany({ where: { costItemId } });
  const materialCostAgg = details.reduce((sum, d) => sum + resolveDetailCost(d), 0);
  const laborCostAgg = details.reduce((sum, d) => sum + Number(d.laborCost ?? 0), 0);
  const otherCostAgg = details.reduce((sum, d) => sum + Number(d.otherCost ?? 0), 0);
  await db.costItem.update({
    where: { id: costItemId },
    data: { materialCostAgg, laborCostAgg, otherCostAgg },
  });
}
```

**src/middleware.ts** - Next.js middleware at the project root (src/):

```typescript
export function middleware(request: NextRequest) {
  const sessionCookie = getSessionCookie(request);
  if (!sessionCookie) {
    return NextResponse.redirect(new URL("/login", request.url));
  }
  return NextResponse.next();
}

export const config = {
  matcher: ["/projects/:path*", "/dashboard/:path*"],
};
```

This middleware:
- Runs on all `/projects/*` and `/dashboard/*` routes (the protected app pages)
- Checks for a valid session cookie using Better Auth's helper
- Redirects to `/login` if no session cookie found
- Allows authenticated requests to proceed

Per AUTH-04: Protected routes redirect unauthenticated users to login.
Per AUTH-05: tRPC-level role checks handle project access control (done in routers above).

NOTE: This is a cookie-presence check, not a full session validation. Full validation happens in the tRPC context when `auth.api.getSession()` verifies the session against the database. The middleware is a fast redirect for unauthenticated users visiting protected pages.
